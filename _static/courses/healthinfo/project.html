import pandas as pd
from synthea import Generator, GeneratorOptions

# 2. ตั้งค่า GeneratorOptions โหลดไฟล์ประชากรไทย thai_demographics.csv
options = GeneratorOptions()
options.population_size = 1000
options.seed = 42
options.demographics_file = "thai_demographics.csv"  # โหลดไฟล์โครงสร้าง 77 จังหวัด
options.state = "Thailand"
options.city = "อำนาจเจริญ"                 # ระบุจังหวัดเป้าหมาย (หรือสุ่มจาก csv)

# 3. รันสร้างข้อมูลประชากรผู้ป่วยสังเคราะห์ด้วย PySynthea Engine
generator = Generator(options)
cohort = generator.run()

# 4. แปลงผลลัพธ์ลง Pandas DataFrame และส่งออก CSV
df_raw = cohort.to_dataframe(table="patients")
df_raw.to_csv('project_raw_patients.csv', index=False)

print("✅ บันทึก project_raw_patients.csv สำเร็จโดยใช้ PySynthea + thai_demographics.csv!")

ขั้นตอนที่ 3: การตรวจตรรกะเวลาและค่าที่เป็นไปไม่ได้ทางคลินิก

แปลงข้อความวันที่เป็น `datetime` เพื่อตรวจสอบเงื่อนไขความขัดแย้งเชิงเวลา (`dischtime >= admittime`) และกรองค่าที่เป็นไปไม่ได้ออก

💻 Python Script: temporal_validation.py Ready to Copy-and-Paste 
import pandas as pd

df = pd.read_csv('project_raw_patients.csv')

# 1. แปลงคอลัมน์ข้อความวันที่ให้อยู่ในรูป datetime object
df['admittime'] = pd.to_datetime(df['admittime'])
df['dischtime'] = pd.to_datetime(df['dischtime'])

# 2. คำนวณระยะเวลาการนอนโรงพยาบาล (Length of Stay - LOS) ในหน่วยวัน
df['los_days'] = (df['dischtime'] - df['admittime']).dt.total_seconds() / 86400

# 3. กรองเฉพาะแถวข้อมูลที่มีตรรกะเวลาถูกต้อง (los_days >= 0)
valid_time_mask = df['los_days'] >= 0
df_time_clean = df[valid_time_mask].copy()

# 4. กรองค่าสัญญานชีพที่เป็นไปไม่ได้ทางคลินิก (Physiologically Implausible Bounds)
# ค่า SBP ปกติของมนุษย์ควรอยู่ระหว่าง 50 ถึง 300 mmHg
valid_vitals_mask = (df_time_clean['sbp'].isnull()) | (df_time_clean['sbp'].between(50, 300))
df_valid = df_time_clean[valid_vitals_mask].copy()

print(f"จำนานแถวเดิม: {len(df)} ➔ หลังตรวจตรรกะเวลาและค่าเป็นไปไม่ได้: {len(df_valid)}")

ขั้นตอนที่ 4: การตัด Outliers ด้วยวิธี IQR และการส่งออกข้อมูล

ใช้ค่าพิสัยระหว่างควอร์ไทล์ (IQR) ในการตัด Outliers ทางสถิติ และส่งออกไฟล์ CSV สะอาดพร้อมสำหรับขั้นตอนการคุ้มครองความเป็นส่วนตัว (PDPA) ในชั่วโมงถัดไป

💻 Python Script: iqr_clean_export.py Ready to Copy-and-Paste 
import pandas as pd

# โหลดชุดข้อมูลที่ผ่านการตรวจตรรกะเวลาแล้ว
df = df_valid.copy()

# 1. คำนวณค่า Quantile Q1 (25%), Q3 (75%) และ IQR สำหรับคอลัมน์ sbp
Q1 = df['sbp'].quantile(0.25)
Q3 = df['sbp'].quantile(0.75)
IQR = Q3 - Q1

lower_bound = Q1 - 1.5 * IQR
upper_bound = Q3 + 1.5 * IQR

# 2. กรองข้อมูลเฉพาะแถวที่อยู่ในช่วง [Lower Bound, Upper Bound] หรือเป็นค่าว่าง (Null)
iqr_mask = (df['sbp'].isnull()) | ((df['sbp'] >= lower_bound) & (df['sbp'] <= upper_bound))
df_cleaned = df[iqr_mask].copy()

# 3. ส่งออกชุดข้อมูลทำความสะอาดแล้วลงไฟล์ CSV (ระบุ index=False เพื่อคงโครงสร้างคอลัมน์)
df_cleaned.to_csv('cleaned_patients_hour1.csv', index=False)

print(f"ขอบเขต IQR ของ SBP: [{lower_bound:.1f}, {upper_bound:.1f}] mmHg")
print(f"✅ บันทึกไฟล์ cleaned_patients_hour1.csv เรียบร้อยแล้ว (คงเหลือ {len(df_cleaned)} แถว)")

Slide 7: Legal Frameworks AMMS 302 | Hour 2

เปรียบเทียบกรอบกฎหมาย: พ.ร.บ. คุ้มครองข้อมูลส่วนบุคคล (PDPA) vs. HIPAA

ข้อมูลสุขภาพจัดเป็น **ข้อมูลส่วนบุคคลอ่อนไหว (Sensitive Personal Data)** ซึ่งต้องมีมาตรการคุ้มครองและกำกับดูแลขั้นสูงสุดตามกฎหมายทั้งในระดับประเทศและสากล

🇹🇭

Thailand PDPA พ.ศ. 2562 (มาตรา 26)

  • Explicit Consent: การเก็บรวบรวมข้อมูลสุขภาพต้องได้รับความยินยอมโดยชัดแจ้ง
  • ข้อยกเว้นทางการแพทย์: ยกเว้นขอ consent เมื่อเป็นประโยชน์ต่อชีวิต (Vital Interest), การรักษาพยาบาลโดยผู้ประกอบวิชาชีพ, หรือสาธารณสุข
  • ผู้มีส่วนได้ส่วนเสีย: Data Subject (คนไข้), Data Controller (รพ.), Data Processor (คลาวด์/ผู้ประมวลผล)
  • บทลงโทษ: โทษปรับทางปกครองสูงสุด 5 ล้านบาท และโทษทางอาญา
🇺🇸

US HIPAA (Privacy & Security Rule)

  • PHI Protection: คุ้มครอง Protected Health Information ในทุกรูปแบบ
  • Minimum Necessary Standard: เข้าถึงข้อมูลเฉพาะเท่าที่จำเป็นต่อการปฏิบัติงาน
  • 2 วิธีการ De-identification:
    1. Safe Harbor Method: ตัดข้อมูลระบุตัวตน 18 รายการ
    2. Expert Determination: ใช้หลักสถิติมั่นใจว่าความเสี่ยงชี้ตัวต่ำมาก
  • มาตรฐานอ้างอิง: ใช้กับชุดข้อมูลวิจัยเปิดระดับโลก เช่น MIMIC-IV
📌 หลักการสำคัญ: ข้อมูลส่วนบุคคลที่ผ่านกระบวนการ De-identification จนไม่สามารถระบุตัวตนได้อีกต่อไป จะหลุดพ้นจากขอบเขตบังคับของ PDPA
Slide 8: Safe Harbor 18 Identifiers AMMS 302 | Hour 2

หลักเกณฑ์ Safe Harbor: 18 รายการข้อมูลระบุตัวตนที่ต้องกำจัดออก

ตามมาตรฐาน HIPAA การนำชุดข้อมูลสุขภาพไปใช้ในการวิจัยหรือพัฒนาระบบ AI จะต้องทำการลบหรือแปลงข้อมูลระบุตัวตนทั้ง 18 รายการดังต่อไปนี้:

1. ชื่อ-นามสกุล (Names)
2. ที่อยู่อาศัยย่อยกว่าจังหวัด (Geographic data)
3. วันที่เกี่ยวกับบุคคล (ยกเว้นปี ค.ศ.)
4. หมายเลขโทรศัพท์ (Phone)
5. หมายเลขแฟกซ์ (Fax)
6. ที่อยู่อีเมล (Email)
7. เลขบัตรประชาชน / SSN
8. เลขประจำตัวผู้ป่วย (HN / MRN)
9. เลขที่สิทธิประกันสุขภาพ
10. เลขที่บัญชีธนาคาร
11. เลขใบอนุญาต / ทะเบียนรถ
12. Serial Number เครื่องมือแพทย์
13. Web URL ที่ระบุบุคคล
14. IP Address
15. อัตลักษณ์ชีวภาพ (Biometrics)
16. ภาพถ่ายใบหน้าเต็ม (Full-face photo)
17. รหัสพันธุกรรม / ลายนิ้วมือ
18. หมายเลขระบุตัวตนเฉพาะอื่นใด
⚠️
กฎการรวมกลุ่มอายุเกิน 89 ปี (Age > 89 Rule):
เนื่องจากผู้สูงอายุที่มีอายุเกิน 89 ปีจัดเป็นกลุ่มประชากรขนาดเล็กที่มีความเสี่ยงสูงต่อการถูกชี้ตัว (Extreme Outliers) ตามหลัก HIPAA จึงต้องแปลงอายุที่มากกว่า 89 ปีทั้งหมดให้เป็นหมวดหมู่รวม เช่น '90+'
Slide 9: Linkage Attack & Salt-Hashing AMMS 302 | Hour 2

ความเสี่ยง Linkage Attack และการเข้ารหัสด้วย SHA-256 + Salt

Linkage Attack (Sweeney, 1997): แม้จะลบชื่อออก แต่เพียงใช้ รหัสไปรษณีย์ + วันเกิด + เพศ สามารถระบุตัวตนประชากรในสหรัฐฯ ได้ถูกต้องถึง **87%** เมื่อนำไปเทียบตารางทะเบียนราษฎร์

💻 Python Script: salt_hashing_demo.py Ready to Copy-and-Paste 
import hashlib

# 1. กำหนดค่า Secret Salt (ต้องเก็บรักษาสมุดรหัสนี้ไว้ในที่ปลอดภัย ไม่ commit ขึ้น GitHub)
SALT = "AMMS302_HEALTH_INFORMATICS_SALT_KEY_2026"

def pseudonymize_id(original_id):
    """
    แปลงรหัสเวชระเบียน (HN) เป็น Pseudonym Hash สั้น 16 หลักด้วย SHA-256 + Salt
    เพื่อป้องกัน Rainbow Table Attacks และการสืบค้นย้อนกลับโดยไม่มี Secret Key
    """
    salted_input = SALT + str(original_id)
    hash_object = hashlib.sha256(salted_input.encode('utf-8'))
    return hash_object.hexdigest()[:16]

# ทดสอบการแปลงรหัสผู้ป่วย HN-00001
raw_hn = "HN-00001"
hashed_hn = pseudonymize_id(raw_hn)

print(f"รหัสเดิม (Direct Identifier): {raw_hn}")
print(f"รหัสใหม่ (Pseudonymized Hash): {hashed_hn}")
# ผลลัพธ์: e9c08d5f3a21b84e (แปลงกลับไม่ได้ถ้าไม่มี Secret Salt)

def hash_patient_id(patient_id):
    return hashlib.sha256((SALT + str(patient_id)).encode('utf-8')).hexdigest()[:16]

# 3. แปลงรหัสประจำตัวด้วย Pseudonymization
df['patient_id_hash'] = df['patient_id'].apply(hash_patient_id)

# 4. ทำ Age Binning และใช้กฎ Age > 89 Rule
def bin_age(age):
    if age >= 90:
        return '90+' # HIPAA Outlier Rule
    group_start = (age // 10) * 10
    return f"{group_start}-{group_start+9}"

df['age_group'] = df['age'].apply(bin_age)

# 5. ลบข้อมูลระบุตัวตนโดยตรง (Direct Identifiers) ทั้งหมดตาม HIPAA Safe Harbor
df_deidentified = df.drop(columns=['patient_id', 'patient_name', 'national_id', 'age'])

# 6. ส่งออกชุดข้อมูล De-identified พร้อมสำหรับนำไปออกแบบฐานข้อมูลในชั่วโมงที่ 3
df_deidentified.to_csv('deidentified_patients_hour2.csv', index=False)

print("✅ สร้างไฟล์ deidentified_patients_hour2.csv เรียบร้อยแล้ว!")

Slide 12: OLTP vs OLAP & Normalization AMMS 302 | Hour 3

สถาปัตยกรรมฐานข้อมูลและขั้นตอนการแตกตาราง (Normalization 1NF ➔ 3NF)

การนำตารางแบน (Flat Spreadsheet) มาจัดเก็บโดยตรงทำให้เกิด **Data Redundancy** และข้อผิดพลาดขณะปรับปรุงข้อมูล (Insertion, Update, Deletion Anomalies)

🔄 OLTP vs. OLAP ในระบบสุขภาพ

  • OLTP (Online Transaction Processing): ระบบ Hospital Information System (HIS) สำหรับการลงบันทึกเวชระเบียนประจำวัน เน้นความรวดเร็วในการ C.R.U.D. และต้องผ่าน Normalization (3NF) เพื่อป้องกันข้อมูลขัดแย้ง
  • OLAP (Online Analytical Processing): ระบบคลังข้อมูล (Data Warehouse / DuckDB) สำหรับวิเคราะห์เชิงลึก เน้นการอ่านข้อมูลปริมาณมาก (Read-Optimized)

📐 ลำดับการทำ Normalization (1NF ➔ 3NF)

  • 1NF (First Normal Form): ทุกเซลล์ต้องเป็นค่าเดี่ยว (Atomic Values) ไม่มี Repeating Groups
  • 2NF (Second Normal Form): อยู่ใน 1NF และทุกคอลัมน์ที่ไม่ใช่ PK ต้องขึ้นตรงกับ Primary Key แบบเต็มสูบ (Full Functional Dependency)
  • 3NF (Third Normal Form): อยู่ใน 2NF และไม่มีการพึ่งพาแบบส่งผ่าน (No Transitive Dependency: ข้อมูลที่ไม่ใช่ Key ห้ามขึ้นกับข้อมูลที่ไม่ใช่ Key ด้วยกัน)
⚠️ ตัวอย่าง Anomaly: หากเก็บชื่อยาและขนาดไว้ในตารางผู้ป่วยโดยตรง เมื่อมีการแก้ไขชื่อยา แพทย์ต้องตามแก้ทุกแถวที่มีผู้ป่วยรับยานั้น (Update Anomaly)

ขั้นตอนที่ 1: การจำแนกชั้นความเสี่ยงด้วย `CASE WHEN` ตามแนวทางเวชปฏิบัติ

💻 SQL Query: cdss_risk_triage.sql Ready to Copy-and-Paste 
SELECT 
    patient_id_hash,
    sbp,
    hba1c,
    lab_date,
    CASE
        WHEN sbp >= 180 OR hba1c >= 9.0 THEN 'CRITICAL'   -- วิกฤต: เสี่ยงภาวะแทรกซ้อนเฉียบพลัน
        WHEN sbp >= 140 OR hba1c >= 7.0 THEN 'HIGH'       -- สูง: ควบคุมไม่ได้ตามเป้าหมาย
        WHEN sbp >= 120 OR hba1c >= 5.7 THEN 'MODERATE'   -- ปานกลาง: ระยะเสี่ยง (Pre-HT / Pre-DM)
        ELSE 'NORMAL'                                    -- ปกติ
    END AS risk_tier
FROM vitals_labs
ORDER BY sbp DESC;
Slide 18: Anti-Join Pattern AMMS 302 | Hour 4

ขั้นตอนที่ 2: การค้นหากลุ่มเสี่ยงวิกฤตที่ไม่ได้รับยาด้วย Anti-Join (`NOT EXISTS`)

ค้นหาผู้ป่วยที่มีค่าสัญญาณชีพวิกฤต (SBP >= 180) แต่ **"ไม่มีประวัติการสั่งจ่ายยา"** ในตาราง prescriptions (หลีกเลี่ยงกับดัก NULL ของ `NOT IN`)

💻 SQL Query: care_gap_anti_join.sql Ready to Copy-and-Paste 
SELECT 
    v.patient_id_hash,
    v.sbp,
    v.hba1c,
    v.lab_date
FROM vitals_labs v
WHERE v.sbp >= 180
  -- รูปแบบ Anti-Join Pattern: ค้นหาผู้ป่วยที่ "ไม่พบข้อมูล" ในตารางใบสั่งยา
  AND NOT EXISTS (
      SELECT 1 
      FROM prescriptions p
      WHERE p.patient_id_hash = v.patient_id_hash
  );
⚠️
คำเตือนเรื่อง SQL Null Trap: ห้ามใช้ WHERE id NOT IN (SELECT id FROM ...) เพราะหากคอลัมน์ใน Subquery มีค่า NULL แม้แต่แถวเดียว ผลลัพธ์ทั้งหมดจะกลายเป็นเซตว่างทันที!
Slide 19: Window Functions AMMS 302 | Hour 4

ขั้นตอนที่ 3: การดึงสัญญาณชีพล่าสุดด้วย Window Functions (`ROW_NUMBER`)

ใช้ ROW_NUMBER() OVER (PARTITION BY ... ORDER BY ...) เพื่อดึงเฉพาะผลแล็บบันทึกล่าสุดของผู้ป่วยแต่ละคน

💻 SQL Query: latest_vitals_window.sql Ready to Copy-and-Paste 
-- ใช้ Common Table Expression (CTE) ร่วมกับ Window Function
WITH RankedVitals AS (
    SELECT 
        patient_id_hash,
        sbp,
        hba1c,
        lab_date,
        -- จัดลำดับประวัติตามวันที่ล่าสุดของผู้ป่วยแต่ละราย (rn = 1 คือพารามิเตอร์ล่าสุด)
        ROW_NUMBER() OVER (
            PARTITION BY patient_id_hash 
            ORDER BY lab_date DESC
        ) AS rn
    FROM vitals_labs
)
SELECT 
    patient_id_hash,
    sbp,
    hba1c,
    lab_date AS latest_visit_date
FROM RankedVitals
WHERE rn = 1;

💻 Python Script: run_cdss_engine.py Ready to Copy-and-Paste 
import sqlite3
import pandas as pd

# 1. เชื่อมต่อฐานข้อมูล SQLite ที่สร้างไว้จากชั่วโมงที่ 3
conn = sqlite3.connect('health_cdss.db')

# 2. คำสั่ง SQL รวมพลัง: CTE + Window Function + Anti-Join + CASE WHEN Risk Triage
cdss_query = """
WITH LatestVitals AS (
    SELECT 
        patient_id_hash, sbp, hba1c, lab_date,
        ROW_NUMBER() OVER (PARTITION BY patient_id_hash ORDER BY lab_date DESC) as rn
    FROM vitals_labs
)
SELECT 
    v.patient_id_hash,
    p.age_group,
    p.gender,
    v.sbp,
    v.hba1c,
    v.lab_date,
    CASE 
        WHEN v.sbp >= 180 OR v.hba1c >= 9.0 THEN 'CRITICAL'
        WHEN v.sbp >= 140 OR v.hba1c >= 7.0 THEN 'HIGH'
        ELSE 'MODERATE'
    END AS risk_tier
FROM LatestVitals v
JOIN patients p ON v.patient_id_hash = p.patient_id_hash
WHERE v.rn = 1
  AND NOT EXISTS (
      SELECT 1 FROM prescriptions rx 
      WHERE rx.patient_id_hash = v.patient_id_hash
  )
ORDER BY v.sbp DESC;
"""

# 3. ดึงผลลัพธ์ใส่ Pandas DataFrame
df_triage = pd.read_sql_query(cdss_query, conn)
conn.close()

# 4. ส่งออกรายงานผู้ป่วยเสี่ยงวิกฤตขาดการรับยา (Care Gap Report)
df_triage.to_csv('critical_triage_report.csv', index=False)

print("🚨 รายงานผู้ป่วยกลุ่มวิกฤตขาดการรับยา (Care Gap Triage Queue):")
print(df_triage.head(5))

import sqlite3
import pandas as pd
from django.shortcuts import render

def clinician_dashboard(request):
    """View สำหรับประมวลผลและแสดงรายการผู้ป่วยกลุ่มเสี่ยงวิกฤตที่ขาดการรับยา"""
    conn = sqlite3.connect('health_cdss.db')
    
    cdss_query = """
    WITH LatestVitals AS (
        SELECT patient_id_hash, sbp, hba1c, lab_date,
               ROW_NUMBER() OVER (PARTITION BY patient_id_hash ORDER BY lab_date DESC) as rn
        FROM vitals_labs
    )
    SELECT v.patient_id_hash, p.age_group, p.gender, v.sbp, v.hba1c, v.lab_date,
           CASE 
               WHEN v.sbp >= 180 OR v.hba1c >= 9.0 THEN 'CRITICAL'

               ELSE 'MODERATE'
           END AS risk_tier

Slide 30: 3P Safety Audit & Wrap-up AMMS 302 | Hour 6

ขั้นตอนสุดท้าย: การประเมินความปลอดภัย 3P Safety Audit และบทสรุปภาพรวม

ตรวจสอบความสมบูรณ์ของระบบสุขภาพดิจิทัลตามกรอบความปลอดภัย 3 ด้าน ก่อนนำเสนอผลงานฉบับสมบูรณ์

1. Patient Safety (ความปลอดภัยผู้ป่วย)

ระบบ CDSS มีเกณฑ์ Cut-off แม่นยำตามแนวทางเวชปฏิบัติ แจ้งเตือน Care Gap ผู้ป่วยวิกฤตที่ขาดการรับยาได้ถูกต้อง ลดความผิดพลาดทางการแพทย์

2. Personnel Safety (ความปลอดภัยเจ้าหน้าที่)

ออกแบบแดชบอร์ด Triage Queue ให้อ่านง่าย ลดภาระงานแบบภาระทางสมอง (Cognitive Overload) ของบุคลากรทางการแพทย์

3. Public Safety & PDPA (ความปลอดภัยสาธารณะ)

ปกปิดตัวตนด้วย Salt-Hashing และ HIPAA Safe Harbor ป้องกันข้อมูลรั่วไหล สอดคล้องกับ พ.ร.บ. คุ้มครองข้อมูลส่วนบุคคล พ.ศ. 2562
